Date Arithmetic — Patterns

# Days between two dates
delta = date2 - date1
days  = delta.days

# Date N days from now
future = date.today() + timedelta(days=N)

# Date N days ago
past = date.today() - timedelta(days=N)

# Day number in the 1825 system
n = (target_date - START_DATE).days + 1

# Calendar date for day number n
d = START_DATE + timedelta(days=n - 1)

# How many days until a deadline?
diff = target - date.today()
print(f'{diff.days} days remaining')

# Is a deadline in the past?
